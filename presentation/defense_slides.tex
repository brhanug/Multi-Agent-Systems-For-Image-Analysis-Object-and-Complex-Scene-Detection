\begin{frame}{Core Methodology: Scene-Aware Agreement \& Uncertainty}
\small
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Scene-Aware Agreement (SAA) Components}
\begin{itemize}
  \item \textbf{Object Grounding ($A_i^{object}$)}: Overlap of spatial candidate bounding boxes:
    \[ A_i^{object} = \operatorname{IoU}(B_i^{YOLO}, B_i^{VLM}) \]
  \item \textbf{Scene Agreement ($A_i^{scene}$)}: Matches zero-shot classifier and semantic VLM scene type:
    \[ A_i^{scene} = \mathbb{1}[\hat{s}_i^{CLIP} = \hat{s}_i^{VLM}] \]
  \item \textbf{Semantic Similarity ($A_i^{vlm}$)}: Contextual taxonomic alignment:
    \[ A_i^{vlm} = \cos(\mathbf{e}^{caption}_i, \mathbf{e}^{taxonomy}_i) \]
\end{itemize}
\column{0.48\textwidth}
\begin{block}{Scene-Aware Agreement (SAA) Equation}
\[
SAA_i = w_o A_i^{object} + w_s A_i^{scene} + w_v A_i^{vlm}
\]
\end{block}
\begin{block}{Epistemic Uncertainty \& Routing}
Epistemic uncertainty ($U_i$) is the inverse of multi-agent agreement. Fused with Scene Complexity ($SCI_i$), it drives the triage queue:
\[
U_i = 1 - SAA_i
\]
\[
\text{ReviewPriority}_i = \alpha U_i + \beta SCI_i
\]
\end{block}
\end{columns}
\end{frame}

We model the probability that a generated label is incorrect using agreement and complexity features:
\[ P(y_i = 1 \mid \mathbf{x}_i) = \sigma(z_i) \]

\begin{frame}{Appendix: Scene Complexity Index (SCI) Details}
\small
We formally model visual and consensus difficulty with the Scene Complexity Index:
\begin{block}{Scene Complexity Index ($SCI_i$)}
\[
SCI_i = \alpha_1 D_i + \alpha_2 O_i + \alpha_3 L_i + \alpha_4 C_i
\]
\end{block}
\textbf{Detailed Scientific Variables}:
\begin{itemize}
  \item \textbf{Object Density ($D_i$)}: Normalized count of candidate bounding boxes. Extreme visual clutter increases detection failure rates.
  \item \textbf{Occlusion Score ($O_i$)}: Total overlapping area between candidate boxes. Occlusion/overlap degrades spatial grounding accuracy.
  \item \textbf{Low-Contrast Indicator ($L_i$)}: Visual entropy and fading metrics. Models chemical decay, grain, and contrast fading in archival photos.
  \item \textbf{Cross-Agent Conflict ($C_i$)}: Standard deviation $\bar{\sigma}_{\text{disagree}}$ across the agent ensemble, capturing irreducible semantic ambiguity.
\end{itemize}
\begin{block}{Complexity Mediation}
As scene complexity increases, the monolithic detector collapses. Complexity positively mediates the multi-agent advantage (Spearman $r_s = 0.575$).
\end{block}
\end{frame}

\begin{frame}{Appendix: Why Single-Model Metrics Are Not Enough}
\begin{columns}[T]
\column{0.50\textwidth}
\begin{alertblock}{Core Discovery}
A standalone spatial detector is structurally blind to its own collapse under domain shift.
\end{alertblock}
\begin{itemize}
  \item Proxy mAP50: \textbf{0.989} (deceptive).
  \item Detector--VLM misalignment: \textbf{93.0\%} of all images.
  \item \textbf{YOLO Role}: The detector acts as a \textbf{candidate proposal generator} rather than a trusted recognition system.

\begin{frame}{Appendix: Threats to Validity}
\small
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Internal Validity}
\textbf{Pseudo-label dependence}: The YOLOv11 student model is distilled from foundational VLMs, meaning it initially inherits their web-scale semantic biases and systematic zero-shot noise before active learning correction.
\end{block}
\begin{block}{External Validity}
\textbf{Archive \& Benchmark Scale}: The evaluation is restricted to the Colibri archive, and the perfect Precision@10 is based on 10 curated queries; generalisation to unrestricted open-world searches remains untested.
\end{block}

\column{0.48\textwidth}
\begin{block}{Construct Validity}
\textbf{SAA Operationalization}: Agreement is modeled as a weighted linear combination of spatial, semantic, and scene priors, which may not capture all latent aspects of human interpretive ambiguity.
\end{block}
\begin{block}{Statistical Conclusion Validity}
\textbf{Extreme class imbalance}: $77.9\%$ of verified images belong to the Drawing scene category, which may restrict the statistical generalisability of AWLF calibration to rare categories.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Appendix: Key Limitations to State Transparently}
\begin{itemize}
  \item \textbf{Proxy detector metrics}: final mAP50 is based on pseudo-label validation, not full human ground truth.
  \item \textbf{Class imbalance}: expert subset is dominated by the Drawing scene category ($n=624$ within the 801-image cohort).
  \item \textbf{Small ambiguity cohort}: Reviewed-and-left-blank group has only $n=5$, so statistical power is limited.
  \item \textbf{Heuristic weighting}: SCI and triage coefficients are currently hand-defined, not fully optimised.
  \item \textbf{Modern VLM bias}: web-scale models may impose contemporary semantics on historical subjects.
\end{itemize}
\end{frame}
